revenue LM = 
VAR _CM = [total_revenue]
VAR _PM =
CALCULATE(
    [total_revenue],
    DATEADD('Date'[Date], -1, MONTH)
)
VAR _perc = DIVIDE(_CM - _PM, _PM)
VAR _format =
SWITCH(
    TRUE(),
    _perc > 0, UNICHAR(11165) & " " & FORMAT(_perc, "0.0%"),
    _perc < 0, UNICHAR(11167) & " " & FORMAT(_perc * - 1, "0.0%"),
    FORMAT(_perc, "0.0%")
)
RETURN
_format
